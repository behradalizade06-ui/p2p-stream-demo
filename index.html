// tunnel-video-server.js // // Install once: // npm init -y // npm install ws qrcode // // Run: // node tunnel-video-server.js const http = require("http"); const WebSocket = require("ws"); const QRCode = require("qrcode"); const { spawn } = require("child_process"); const PORT = 8081; const GITHUB_PAGE = "https://behradalizade06-ui.github.io/p2p-stream-demo/"; const ROOM = "demo"; const MAX_CLIENT_BUFFER = 350 * 1024; const server = http.createServer((req, res) => { res.writeHead(200, { "Content-Type": "text/plain" }); res.end("Low latency binary tunnel stream OK\n"); }); const wss = new WebSocket.Server({ server, maxPayload: 30 * 1024 * 1024 }); const rooms = new Map(); let tunnelProcess = null; let tunnelUrlFound = false; function getRoom(roomName) { if (!rooms.has(roomName)) { rooms.set(roomName, { tablet: null, clients: new Set(), streaming: false }); } return rooms.get(roomName); } function sendJson(ws, data) { if (ws && ws.readyState === WebSocket.OPEN) { ws.send(JSON.stringify(data)); } } function broadcastJson(room, data) { const payload = JSON.stringify(data); for (const client of room.clients) { if (client.readyState === WebSocket.OPEN) { client.send(payload); } } } function broadcastBinary(room, buffer) { for (const client of room.clients) { if (client.readyState !== WebSocket.OPEN) { continue; } // IMPORTANT: // Do not allow slow phones to build latency. // If phone/network is slow, skip this frame. if (client.bufferedAmount > MAX_CLIENT_BUFFER) { continue; } client.send(buffer, { binary: true }); } } wss.on("connection", (ws) => { let role = null; let roomName = ROOM; console.log("New WebSocket connection"); ws.on("message", (message, isBinary) => { const room = getRoom(roomName); // Binary messages are video frames from tablet. if (isBinary) { if (role === "tablet" && room.streaming) { broadcastBinary(room, message); } return; } let msg; try { msg = JSON.parse(message.toString()); } catch { return; } if (msg.type === "join") { role = msg.role; roomName = msg.room || ROOM; const joinedRoom = getRoom(roomName); if (role === "tablet") { joinedRoom.tablet = ws; joinedRoom.streaming = false; console.log("Tablet joined room:", roomName); sendJson(ws, { type: "joined", role: "tablet" }); } if (role === "client") { joinedRoom.clients.add(ws); console.log("Client joined room:", roomName); sendJson(ws, { type: "joined", role: "client" }); if (joinedRoom.streaming) { sendJson(ws, { type: "stream-start" }); } } return; } if (msg.type === "stream-start" && role === "tablet") { room.streaming = true; console.log("Stream started:", roomName); broadcastJson(room, { type: "stream-start" }); return; } if (msg.type === "feedback" && role === "client") { if (room.tablet && room.tablet.readyState === WebSocket.OPEN) { sendJson(room.tablet, { type: "feedback", avgDelay: msg.avgDelay, lastDelay: msg.lastDelay, receivedFps: msg.receivedFps }); } return; } if (msg.type === "stream-stop" && role === "tablet") { room.streaming = false; console.log("Stream stopped:", roomName); broadcastJson(room, { type: "stream-stop" }); return; } }); ws.on("close", () => { const room = getRoom(roomName); if (role === "tablet" && room.tablet === ws) { console.log("Tablet disconnected"); room.tablet = null; room.streaming = false; broadcastJson(room, { type: "stream-stop" }); } if (role === "client") { console.log("Client disconnected"); room.clients.delete(ws); } if (!room.tablet && room.clients.size === 0) { rooms.delete(roomName); } }); }); server.listen(PORT, () => { console.log("\nLow latency binary tunnel server running:"); console.log(`http://localhost:${PORT}`); console.log("\nStarting Cloudflare Tunnel..."); tunnelProcess = spawn("cloudflared", [ "tunnel", "--url", `http://localhost:${PORT}` ]); function handleTunnelOutput(data) { const text = data.toString(); process.stdout.write(text); const match = text.match(/https:\/\/[a-zA-Z0-9.-]+\.trycloudflare\.com/); if (match && !tunnelUrlFound) { tunnelUrlFound = true; const httpsTunnel = match[0].replace(/\/$/, ""); const wssTunnel = httpsTunnel.replace("https://", "wss://"); const tabletUrl = GITHUB_PAGE + "?role=tablet" + "&room=" + encodeURIComponent(ROOM) + "&server=" + encodeURIComponent(wssTunnel); const clientUrl = GITHUB_PAGE + "?role=client" + "&room=" + encodeURIComponent(ROOM) + "&server=" + encodeURIComponent(wssTunnel); console.log("\n\n=============================="); console.log("SYSTEM READY"); console.log("=============================="); console.log("\nCloudflare HTTPS tunnel:"); console.log(httpsTunnel); console.log("\nWebSocket tunnel:"); console.log(wssTunnel); console.log("\nOpen this on TABLET:"); console.log(tabletUrl); console.log("\nClient URL:"); console.log(clientUrl); QRCode.toString(clientUrl, { type: "terminal" }, (err, qr) => { if (!err) { console.log("\nCLIENT QR:\n"); console.log(qr); } }); console.log("\nKeep this terminal open.\n"); } } tunnelProcess.stdout.on("data", handleTunnelOutput); tunnelProcess.stderr.on("data", handleTunnelOutput); tunnelProcess.on("error", (err) => { console.error("\nCloudflared failed:"); console.error(err.message); console.error("\nCheck with:"); console.error("cloudflared --version"); }); tunnelProcess.on("close", (code) => { console.log("\nCloudflare tunnel stopped:", code); }); }); process.on("SIGINT", () => { console.log("\nStopping..."); if (tunnelProcess) { tunnelProcess.kill("SIGINT"); } process.exit(); });
